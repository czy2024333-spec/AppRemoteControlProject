cmake_minimum_required(VERSION 3.16)
project(drone_core VERSION 0.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(drone_core STATIC src/drone_core.cpp src/drone_core_3d.cpp)
set_target_properties(drone_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(drone_core PUBLIC include)
add_library(drone_core_c SHARED src/drone_core_c.cpp src/drone_core_3d_c.cpp)
target_link_libraries(drone_core_c PRIVATE drone_core)
target_include_directories(drone_core_c PUBLIC include)
target_compile_definitions(drone_core_c PRIVATE DC_BUILD_DLL INTERFACE DC_USE_DLL)
add_executable(core_tests tests/core_tests.cpp src/drone_core_c.cpp)
target_link_libraries(core_tests PRIVATE drone_core)
add_executable(scheduler_tests tests/scheduler_tests.cpp)
target_link_libraries(scheduler_tests PRIVATE drone_core)
add_executable(core_3d_tests tests/core_3d_tests.cpp src/drone_core_3d_c.cpp)
target_link_libraries(core_3d_tests PRIVATE drone_core)
add_executable(environment_tests tests/environment_tests.cpp src/drone_core_3d_c.cpp)
target_link_libraries(environment_tests PRIVATE drone_core)
add_executable(core_demo examples/core_demo.cpp)
target_link_libraries(core_demo PRIVATE drone_core)
add_executable(c_api_smoke examples/c_api_smoke.c)
target_link_libraries(c_api_smoke PRIVATE drone_core_c)
if(MSVC)
    foreach(target drone_core drone_core_c core_tests core_3d_tests environment_tests scheduler_tests core_demo c_api_smoke)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    endforeach()
else()
    foreach(target drone_core drone_core_c core_tests core_3d_tests environment_tests scheduler_tests core_demo c_api_smoke)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endforeach()
endif()
enable_testing()
add_test(NAME numerical_and_contract_tests COMMAND core_tests)
add_test(NAME frame_scheduler_tests COMMAND scheduler_tests)
add_test(NAME three_dimensional_core COMMAND core_3d_tests)
add_test(NAME environment_and_recovery COMMAND environment_tests)
add_test(NAME fixed_input_demo COMMAND core_demo)
add_test(NAME native_c_dll_consumer COMMAND c_api_smoke)
